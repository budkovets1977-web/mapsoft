# Структура задач веб-сервера.
# Названия задач/групп используется в настройках конфигураций cfg/config.json и
# виртуальных хостов в cfg/vhosts.ini.

crm                             # Корпоративная CRM

map                             # Группа задач связанных с БД МАП

    buh                         # Группа бухгалтерских задач
        buh_desktop
            buhadmin            # Администрирование (бухгалтерия)
            buhservices         # Учёт платных услуг
    equeue                      # Группа задач электронной очереди
        equeueapp               # Панель администратора электронной очереди
        equeuedisplay           # Вызывное табло электронной очереди
        equeueinfobox           # Инфокиоск электронной очереди
        equeueorc               # Пульт оператора электронной очереди
        equeue_api              # Сервис для предоставления данных из базы МАП в сторонние программы (поиск пациентов)
                                # электронной очереди (поиск пациентов)
    infopanel                   # Информационная панель

    med                         # Группа медицинских задач
        ambulance               # Скорая помощь
        ambulancemobile         # Скорая помощь (мобильное)
        clinic                  # Группа задач поликлиники
            clinic_desktop      # Настольные задачи поликлиники
                medadmin            # Администрирование (медицина)
                medboss             # Рабочее место руководителя (медицина)
                medcard             # Амбулаторная карта
                medclinicstats      # Статистика поликлиники
                meddayhospital      # Дневной стационар
                medexaminations     # Диспансеризация, медосмотры
                medfluorography     # Флюорокартотека
                medfunctional       # Функциональная диагностика
                medlab              # Лабораторная диагностика
                medreception        # Регистратура
                medrecipes          # Льготные рецепты
                medsickday          # Учёт временной нетрудоспособности
                medstomatology      # Стоматология
                medvaccinations     # Иммунопрофилактика
                medxray             # Лучевая диагностика
                medinspectrw_api    # Сервис приема медосмотров работников ЖД из
                                    # поликлиник в базу МАП ЦВЭК (централизованная врачебная экспертная комиссия).
                                    # МАПы поликлиник передают планировщиком в данный сервис, который
                                    # синхронизирует в централизованную базу МАП ЦВЭК.
            clinic_mobile       # Мобильные задачи поликлиники
                homevisit           # Визиты врача на дом

        hospital                # Группа задач стационара
            hospital_desktop    # Настольные задачи стационара
                hospitalhistory     # История болезни
                hospitalreception   # Приёмный покой
            hospital_mobile     # Настольные задачи стационара
                mobilehospitaldoctor# Врач стационара
                mobilehospitalnurse # Медсестра стационара

    pmed                        # Группа платной медицины (медцентры)
        pmeddoctor              # МЦ Врач
        pmedeconomist           # МЦ Экономист
        pmedreception           # МЦ Ресепшн
        pmedinsurance           # Страховые услуги

    svodbuh                     # Сводная бухгалтерская отчётность и ГРС
    synevo                      # Заказы Synevo
    talonbox                    # Инфокиоск для заказа талонов и просмотра прейскуранта платных услуг
    talonby                     # Сервис сайта talon.by

patientident                    # Распределение номеров индивидуальных карт медобслуживания
service131                      # Служба 131, сервер и веб-приложение для клиентов без комплекса МАП
    service131_api              # Сервис службы 131 для клиентов с комплексом МАП обеспечивающий заказ и
                                # отмену талонов
